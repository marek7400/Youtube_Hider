    function detectLatinLanguageLexically(normalizedTitle, titleText) {
        if (/[ąćęłńóśźżĄĆĘŁŃÓŚŹŻ]/u.test(titleText)) {
            return null;
        }

        const cleanTokens = normalizedTitle
            .toLowerCase()
            .split(/\s+/)
            .map(w => w.replace(/^[^\p{L}\p{N}]+|[^\p{L}\p{N}]+$/gu, ''))
            .filter(w => w.length > 0);

        if (cleanTokens.length < 2) return null;

        const polishHits = cleanTokens.filter(w => polishStopwords.has(w)).length;
        if (polishHits >= 2) return null;

        const englishHits = cleanTokens.filter(w => englishStopwords.has(w)).length;

        // Szybkie sprawdzanie unikalnych znaków językowych
        if (/[¿¡ñÑ]/u.test(titleText)) {
            return { name: 'Hiszpański', hits: 2, isScandinavian: false };
        }
        if (/[ğĞıİşŞ]/u.test(titleText)) {
            return { name: 'Turecki', hits: 2, isScandinavian: false };
        }
        if (/[ß]/u.test(titleText)) {
            return { name: 'Niemiecki', hits: 2, isScandinavian: false };
        }
        if (/[ðÐþÞ]/u.test(titleText)) {
            return { name: 'Islandzki / Farerski / Staronorweski', hits: 2, isScandinavian: true };
        }
        if (/[åÅæÆøØ]/u.test(titleText)) {
            return { name: 'Skandynawski', hits: 2, isScandinavian: true };
        }

        let bestMatch = null;
        let maxHits = 0;

        for (const langKey of Object.keys(LEXICAL_ANALYZER)) {
            const langData = LEXICAL_ANALYZER[langKey];
            const hits = cleanTokens.filter(w => langData.words.has(w)).length;
            if (hits >= langData.minHits && hits > englishHits && hits > maxHits) {
                maxHits = hits;
                bestMatch = { name: langData.name, hits, isScandinavian: Boolean(langData.isScandinavian) };
            }
        }

        return bestMatch;
    }

    async function initChromeAiDetector() {
        try {
            const DetectorClass = window.LanguageDetector || (typeof self !== 'undefined' ? self.LanguageDetector : null);
            if (!DetectorClass) {
                aiDetectorStatus = 'unsupported';
                return;
            }

            const availability = await DetectorClass.availability();
            if (availability === 'unavailable') {
                aiDetectorStatus = 'unavailable';
                return;
            }

            aiDetectorStatus = availability;
            aiDetectorInstance = await DetectorClass.create();
            addLog('INFO', 'Pomyślnie zainicjowano Chrome AI LanguageDetector.');
        } catch (err) {
            addLog('WARN', 'Błąd inicjalizacji Chrome LanguageDetector', err.message);
        }
    }

    async function detectLanguageWithAi(titleText) {
        if (!aiDetectorInstance || !titleText) return null;
        if (aiDetectionCache.has(titleText)) {
            return aiDetectionCache.get(titleText);
        }

        try {
            const results = await aiDetectorInstance.detect(titleText);
            if (Array.isArray(results) && results.length > 0) {
                const top = results[0];
                aiDetectionCache.set(titleText, top);
                return top;
            }
        } catch (e) {
            addLog('WARN', 'Błąd podczas detekcji języka AI', e.message);
        }
        return null;
    }

    function processAiDetection(item, titleText, rank) {
        detectLanguageWithAi(titleText).then(res => {
            if (!res) return;
            const isScandicAi = allowScandinavian && scandicLanguageCodes.has(res.detectedLanguage);
            if (!isScandicAi && res.detectedLanguage !== 'pl' && res.detectedLanguage !== 'en' && res.confidence >= 0.70) {
                const lName = langNames[res.detectedLanguage] || res.detectedLanguage.toUpperCase();
                applyHidingState(item, `Tytuł obcojęzyczny: ${lName} (AI: ${Math.round(res.confidence * 100)}%)`, rank);
            }
        });
    }

    function normalizeText(text) {
        if (!text) return '';
        return String(text).normalize('NFKC')
                   .replace(/[\u200B-\u200D\uFEFF\u200E\u200F]/g, '')
                   .replace(/\s+/g, ' ')
                   .trim()
                   .toLowerCase();
    }

    function getChannelNames(item) {
        const names = [];
        if (!item) return names;

        try {
            const allLinks = item.querySelectorAll('a');
            allLinks.forEach(link => {
                const href = link.getAttribute('href') || '';
                let path = href;

                if (href.startsWith('http')) {
                    try { path = new URL(href).pathname; } catch(e) {}
                }

                if (path.startsWith('/@') || path.startsWith('/channel/') || path.startsWith('/c/') || path.startsWith('/user/')) {
                    if (path.startsWith('/@')) {
                        const handle = path.split('/')[1].split('?')[0];
                        names.push(handle);
                    }
                    const visibleText = link.innerText || link.textContent;
                    if (visibleText) names.push(visibleText);
                    if (link.getAttribute('title')) names.push(link.getAttribute('title'));
                }
                if (link.id === 'avatar-link' && link.getAttribute('title')) {
                    names.push(link.getAttribute('title'));
                }
            });

            const textNodes = item.querySelectorAll('ytd-channel-name yt-formatted-string, .yt-channel-name-view-model .yt-core-attributed-string, ytd-channel-name, #channel-name, .ytd-channel-name');
            textNodes.forEach(node => {
                if (node.getAttribute('title')) names.push(node.getAttribute('title'));
                const visibleText = node.innerText || node.textContent;
                if (visibleText) names.push(visibleText);
            });

            // Obsługa nowego układu YouTube: ytContentMetadataViewModelMetadataRow (autorzy i współpraca)
            const metadataRows = item.querySelectorAll('.ytContentMetadataViewModelMetadataRow, .yt-content-metadata-view-model__metadata-row');
            metadataRows.forEach(row => {
                const spans = row.querySelectorAll('.ytContentMetadataViewModelMetadataText, .yt-content-metadata-view-model__metadata-text');
                spans.forEach(span => {
                    // Ignorowanie elementów końcowych (zazwyczaj data)
                    if (span.classList.contains('ytContentMetadataViewModelMetadataTextLastPart') || span.classList.contains('yt-content-metadata-view-model__metadata-text--last-part')) return;

                    const aria = (span.getAttribute('aria-label') || '').toLowerCase();
                    if (aria.includes('wyświetl') || aria.includes('view') || aria.includes('temu') || aria.includes('ago') || aria.includes('stream') || aria.includes('live')) return;

                    const txt = (span.textContent || '').replace(/[\u00A0\s]+/g, ' ').trim();
                    if (!txt) return;

                    // Odrzucanie czystych liczników wyświetleń np. "716 tys.", "1.2 mln", "15K"
                    if (/^\d+(?:[.,]\d+)?\s*(?:tys\.?|mln\.?|mld\.?|k|m|b|views|wyświetleń|wyświetlenia)?$/i.test(txt)) return;

                    // Odrzucanie dat publikacji np. "2 dni temu", "3 hours ago"
                    if (/temu|ago|wczoraj|yesterday|na żywo|live|premiera|premiered/i.test(txt)) return;

                    // Odrzucanie wskaźników czasu np. "10 min", "2 godz"
                    if (/^\d+\s*(?:sek|min|godz|dni|dzień|tyg|mies|lat|rok)\b/i.test(txt)) return;

                    // Sprawdzenie czy element nie występuje za ikoną wyświetleń (leading icon)
                    let prev = span.previousElementSibling;
                    let isAfterIcon = false;
                    while (prev) {
                        if (prev.classList.contains('ytContentMetadataViewModelLeadingIcon') || prev.querySelector('.ytContentMetadataViewModelLeadingIcon')) {
                            isAfterIcon = true;
                            break;
                        }
                        prev = prev.previousElementSibling;
                    }
                    if (isAfterIcon) return;

                    names.push(txt);
                });
            });

            const avatars = item.querySelectorAll('img');
            avatars.forEach(img => {
                if (img.getAttribute('alt')) names.push(img.getAttribute('alt'));
            });

            const avatarShapes = item.querySelectorAll('yt-avatar-shape');
            avatarShapes.forEach(shape => {
                const tooltip = shape.getAttribute('data-tooltip-text');
                if (tooltip) names.push(tooltip);
                const ariaLabel = shape.getAttribute('aria-label');
                if (ariaLabel) names.push(ariaLabel);
            });

            if (names.length === 0) {
                const fallbackLinks = item.querySelectorAll('#channel-name a, ytd-channel-name a');
                fallbackLinks.forEach(l => {
                    const txt = l.innerText || l.textContent;
                    if (txt) names.push(txt);
                });
            }
        } catch (e) {
            addLog('ERROR', 'Błąd w getChannelNames', e.message);
        }

        let cleanedNames = [];
        names.forEach(n => {
            if (!n) return;
            let trimmed = String(n)
                .replace(/(Zweryfikowano|Verified|Z oficjalnego kanału wykonawcy|Official Artist Channel)/gi, '')
                .replace(/[\u200B-\u200D\uFEFF\u200E\u200F]/g, '')
                .trim()
                .replace(/\s+/g, ' ');

            if (trimmed.length > 0) {
                let baseName = trimmed.replace(/(?:^|\s+)(?:i\s+jeszcze\s+\d+|and\s+\d+\s+more|i\s+\d+\s+innych|and\s+\d+\s+others?|oraz\s+\d+\s+innych)(?:\s+.*)?$/i, '').trim();

                baseName = baseName.replace(/^(?:i|and|oraz|&|,)\s+/i, '').trim();
                baseName = baseName.replace(/\s+(?:i|and|oraz|&|,)$/i, '').trim();

                cleanedNames.push(trimmed);
                if (baseName !== trimmed && baseName.length > 0) {
                    cleanedNames.push(baseName);
                }

                const parts = baseName.split(/\s+(?:i|and|oraz|&|,)\s+/i);
                if (parts.length > 1) {
                    parts.forEach(p => {
                        const pt = p.trim();
                        if (pt.length > 0) cleanedNames.push(pt);
                    });
                }
            }
        });

        return [...new Set(cleanedNames)];
    }

    function getBestChannelName(item) {
        const names = getChannelNames(item);
        if (!names || names.length === 0) return null;

        const handle = names.find(n => n.startsWith('@'));
        if (handle) return handle;

        const cleanNames = names.filter(n =>
            !n.startsWith('@') &&
            !/(?:^|\s+)(i|and|oraz|&|,)(?:\s+|$)/i.test(n) &&
            !/jeszcze|more|innych|others/i.test(n)
        );
        if (cleanNames.length > 0) {
            return cleanNames[0];
        }

        return names[0];
    }

    function isChannelMatched(extractedNames, targetList) {
        if (!Array.isArray(extractedNames) || !Array.isArray(targetList)) return false;
        if (extractedNames.length === 0 || targetList.length === 0) return false;

        return extractedNames.some(extracted => {
            const e = normalizeText(extracted);
            if (!e) return false;

            return targetList.some(target => {
                const t = normalizeText(target);
                if (!t) return false;

                if (e === t || e === `@${t}` || `@${e}` === t) return true;

                const safeT = t.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
                const regex = new RegExp(`(?:^|[^\\p{L}\\p{N}])${safeT}(?:[^\\p{L}\\p{N}]|$)`, 'iu');
                if (regex.test(e)) return true;

                return false;
            });
        });
    }

    function calculateRankAndGetTime(item) {
        if (item.querySelector('#ad-badge, .ytd-ad-slot-renderer') || item.textContent.toLowerCase().includes('sponsorowane') || item.textContent.toLowerCase().includes('sponsored')) {
            return { rank: -2, fullText: "sponsorowane" };
        }

        let flatMetadata = "";
        const metadataNodes = item.querySelectorAll('#metadata-line, .yt-core-attributed-string, .yt-lockup-metadata-view-model__metadata, .ytLockupMetadataViewModelMetadata, .yt-content-metadata-view-model__metadata-row, #metadata');

        metadataNodes.forEach(node => {
            flatMetadata += " " + node.textContent.replace(/\s+/g, ' ');
        });

        flatMetadata = normalizeText(flatMetadata);

        if (!flatMetadata.includes('temu') && !flatMetadata.includes('ago')) {
            const lines = normalizeText(item.innerText || item.textContent || "").split(/\r?\n/);
            for (const line of lines) {
                if (line.includes('temu') || line.includes('ago')) {
                    flatMetadata += " " + line.replace(/\s+/g, ' ');
                }
            }
        }

        const sortedUnits = Object.keys(timeUnits).sort((a, b) => b.length - a.length);

        for (const unit of sortedUnits) {
            const safeUnit = unit.replace(/\./g, '\\.');
            const regex = new RegExp(`(?:^|[^\\p{L}])(\\d+)\\s*${safeUnit}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
            const match = flatMetadata.match(regex);
            if (match) {
                return { rank: parseInt(match[1], 10) * timeUnits[unit], fullText: flatMetadata };
            }
        }

        for (const unit of sortedUnits) {
            const safeUnit = unit.replace(/\./g, '\\.');
            const regex = new RegExp(`(?:^|[^\\p{L}])${safeUnit}(?=[^\\p{L}]|$).{0,20}(?:temu|ago)`, 'iu');
            const match = flatMetadata.match(regex);
            if (match) {
                return { rank: 1 * timeUnits[unit], fullText: flatMetadata };
            }
        }

        return { rank: -1, fullText: flatMetadata };
    }

    function applyHidingState(item, hideReason, rank) {
        if (hideReason) {
            let useOverlay = false;
            if (showOverlays) {
                if (hideReason === 'Brak informacji o dacie publikacji') {
                    useOverlay = overlaySettings.noDate;
                } else if (hideReason.startsWith('Wskaźnik czasu') || hideReason === 'Film sponsorowany / Reklama') {
                    useOverlay = overlaySettings.time;
                } else if (hideReason.startsWith('Tytuł obcojęzyczny')) {
                    useOverlay = overlaySettings.lang;
                } else if (hideReason.startsWith('Zablokowane słowo')) {
                    useOverlay = overlaySettings.word;
                } else if (hideReason.startsWith('Zablokowany kanał')) {
                    useOverlay = overlaySettings.channel;
                }
            }

            if (useOverlay) {
                item.style.removeProperty('display');
                item.style.setProperty('order', '9999999', 'important');

                if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                    item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                    item.style.setProperty('margin', '0 8px 24px 8px', 'important');
                } else {
                    item.style.setProperty('width', '100%', 'important');
                    item.style.setProperty('margin', '0 0 24px 0', 'important');
                }
                applyOverlay(item, hideReason);
            } else {
                item.style.setProperty('display', 'none', 'important');
                removeOverlay(item);
            }
        } else {
            item.style.removeProperty('display');
            removeOverlay(item);

            const orderVal = rank >= 0 ? rank : 999999;
            item.style.setProperty('order', orderVal, 'important');

            if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                item.style.setProperty('margin', '0 8px 24px 8px', 'important');

                item.style.setProperty('width', '100%', 'important');
                item.style.setProperty('margin', '0 0 24px 0', 'important');
            }
        }
    }

    function processYouTubeHider() {
        try {
            const gridContainer = document.querySelector('ytd-rich-grid-renderer > #contents') || document.querySelector('ytd-rich-grid-renderer #contents');
            let items;

            if (gridContainer) {
                items = gridContainer.querySelectorAll('ytd-rich-item-renderer, ytd-rich-section-renderer, ytd-video-renderer, ytd-grid-video-renderer, ytd-compact-video-renderer');
            } else {
                items = document.querySelectorAll('ytd-rich-item-renderer, ytd-rich-section-renderer, ytd-video-renderer, ytd-grid-video-renderer, ytd-compact-video-renderer');
            }

            if (!items || items.length === 0) {
                return;
            }

            let hiddenCount = 0;

            if (!pluginEnabled) {
                document.documentElement.classList.remove('youtube-hider-active');
                if (gridContainer) {
                    gridContainer.style.removeProperty('display');
                    gridContainer.style.removeProperty('flex-wrap');
                }
                items.forEach(item => {
                    item.style.removeProperty('display');
                    item.style.removeProperty('order');
                    item.style.removeProperty('width');
                    item.style.removeProperty('margin');
                    removeOverlay(item);
                    const btn = item.querySelector('.yt-hider-block-ch-btn');
                    if (btn) btn.remove();
                });
                updateBadgeUI(0);
                return;
            }

            document.documentElement.classList.add('youtube-hider-active');
            if (gridContainer) {
                gridContainer.style.setProperty('display', 'flex', 'important');
                gridContainer.style.setProperty('flex-wrap', 'wrap', 'important');
            }

            const activeKeywords = Object.keys(keywordSettings).filter(k => keywordSettings[k]);

            // Kompilacja granic słów w Unikodzie dla własnych słów blokowanych (zapobiega blokowaniu np. "przepis" dla "pis")
            const customWordsArr = customBlockedWords.split(',')
                .map(w => normalizeText(w))
                .filter(w => w.length > 0);

            const customWordRegexes = customWordsArr.map(word => {
                const safeWord = word.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
                return {
                    rawWord: word,
                    regex: new RegExp(`(?<![\\p{L}\\p{N}])${safeWord}(?![\\p{L}\\p{N}])`, 'iu')
                };
            });

            items.forEach(item => {
                injectBlockButton(item);

                if (item.dataset.ytHiderUnblocked === "true") {
                    item.style.removeProperty('display');
                    removeOverlay(item);

                    const { rank } = calculateRankAndGetTime(item);
                    const orderVal = rank >= 0 ? rank : 999999;
                    item.style.setProperty('order', orderVal, 'important');

                    if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                        item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                        item.style.setProperty('margin', '0 8px 24px 8px', 'important');
                    } else {
                        item.style.setProperty('width', '100%', 'important');
                        item.style.setProperty('margin', '0 0 24px 0', 'important');
                    }
                    return;
                }

                if (item.querySelector('a[href^="/shorts/"]')) {
                    item.style.setProperty('display', 'none', 'important');
                    return;
                }

                const { rank, fullText } = calculateRankAndGetTime(item);
                let hideReason = null;
                let titleText = '';

                // Pobieranie tytułu bezpośrednio z h3, linku lub span
                const headingEl = item.querySelector('.ytLockupMetadataViewModelHeadingReset, h3[title], h3');
                if (headingEl && headingEl.getAttribute('title')) {
                    titleText = headingEl.getAttribute('title');
                } else {
                    const titleLinkEl = item.querySelector('.ytLockupMetadataViewModelTitle, #video-title, #video-title-link, .yt-lockup-metadata-view-model__title, h3 a');
                    if (titleLinkEl) {
                        titleText = titleLinkEl.getAttribute('title') || titleLinkEl.textContent || titleLinkEl.getAttribute('aria-label') || '';
                    } else if (headingEl) {
                        titleText = headingEl.textContent || '';
                    }
                }
                if (!titleText) {
                    const fallbackSpan = item.querySelector('.ytAttributedStringHost, #video-title');
                    if (fallbackSpan) {
                        titleText = fallbackSpan.textContent || '';
                    }
                }
                if (titleText) titleText = titleText.trim();

                const itemChannelNames = getChannelNames(item);

                if (!hideReason && blockedChannels.length > 0 && itemChannelNames.length > 0) {
                    const isBlocked = isChannelMatched(itemChannelNames, blockedChannels);
                    if (isBlocked) {
                        const matchedCh = itemChannelNames.find(n => isChannelMatched([n], blockedChannels)) || 'Czarna lista';
                        hideReason = `Zablokowany kanał: ${matchedCh}`;
                    }
                }

                if (!hideReason && customWordRegexes.length > 0) {
                    const lowerTitle = normalizeText(titleText);
                    const matched = customWordRegexes.find(entry => {
                        return entry.regex.test(lowerTitle) || itemChannelNames.some(ch => entry.regex.test(normalizeText(ch)));
                    });
                    if (matched) {
                        hideReason = `Zablokowane słowo: "${matched.rawWord}"`;
                    }
                }

                if (blockNonPlEn && !hideReason) {
                    const isChannelAllowed = isChannelMatched(itemChannelNames, allowedChannels);

                    if (!isChannelAllowed && titleText) {
                        const normalizedTitle = titleText.normalize('NFKC');
                        const hasPolishDiacritics = /[ąćęłńóśźżĄĆĘŁŃÓŚŹŻ]/u.test(titleText);

                        if (foreignRegex.test(normalizedTitle)) {
                            // 1. Pisma azjatyckie (chiński, japoński, koreański, tajski)
                            const asianMatches = normalizedTitle.match(/[\p{Script=Han}\p{Script=Hiragana}\p{Script=Katakana}\p{Script=Hangul}\p{Script=Thai}]/gu) || [];

                            // 2. Cyrylica
                            const cyrillicMatches = normalizedTitle.match(/[\p{Script=Cyrillic}]/gu) || [];

                            // 3. Pozostałe alfabety niełacińskie
                            const otherNonLatinMatches = normalizedTitle.match(/[\p{Script=Arabic}\p{Script=Devanagari}\p{Script=Hebrew}\p{Script=Greek}]/gu) || [];

                            // 4. Słowa łacińskie
                            const latinWords = normalizedTitle
                                .split(/\s+/)
                                .map(w => w.replace(/^[^\p{L}\p{N}]+|[^\p{L}\p{N}]+$/gu, ''))
                                .filter(w => /[a-zA-ZąćęłńóśźżĄĆĘŁŃÓŚŹŻ]/i.test(w));

                            // 5. Specyficzne znaki tureckie (ğĞıİşŞ)
                            const turkishCharMatches = normalizedTitle.match(/[ğĞıİşŞ]/gu) || [];

                            // 6. Słowa z obcymi znakami diakrytycznymi (np. niemieckie, francuskie, czeskie)
                            const foreignAccentWords = normalizedTitle
                                .split(/\s+/)
                                .map(w => w.replace(/^[^\p{L}\p{N}]+|[^\p{L}\p{N}]+$/gu, ''))
                                .filter(w => {
                                    if (!foreignRegex.test(w)) return false;
                                    if (/[\p{Script=Han}\p{Script=Hiragana}\p{Script=Katakana}\p{Script=Hangul}\p{Script=Thai}\p{Script=Cyrillic}\p{Script=Arabic}\p{Script=Devanagari}\p{Script=Hebrew}\p{Script=Greek}]/u.test(w)) return false;

                                    // Gdy zezwolono na języki skandynawskie, litery å, æ, ø, ð, þ nie są traktowane jako obce akcenty
                                    if (allowScandinavian) {
                                        const withoutScandic = w.replace(/[åÅæÆøØðÐþÞ]/gu, '');
                                        if (!foreignRegex.test(withoutScandic)) return false;
                                    }
                                    return true;
                                });

                            let shouldBlock = false;
                            let detectedLang = '';

                            if (asianMatches.length > 2) {
                                if (latinWords.length < 3 || asianMatches.length >= 8 || asianMatches.length > latinWords.length * 2) {
                                    shouldBlock = true;

                                    if (/[\p{Script=Hiragana}\p{Script=Katakana}]/u.test(normalizedTitle)) {
                                        detectedLang = `Japoński (${asianMatches.length} znaków Kana/Kanji)`;
                                    } else if (/[\p{Script=Hangul}]/u.test(normalizedTitle)) {
                                        detectedLang = `Koreański (${asianMatches.length} znaków Hangul)`;
                                    } else if (/[\p{Script=Thai}]/u.test(normalizedTitle)) {
                                        detectedLang = `Tajski (${asianMatches.length} znaków)`;
                                    } else {
                                        detectedLang = `Chiński / Azjatycki (${asianMatches.length} znaków Hanzi)`;
                                    }
                                }
                            } else if (cyrillicMatches.length > 3) {
                                if (latinWords.length < 3 || cyrillicMatches.length >= 10 || cyrillicMatches.length > latinWords.length * 2) {
                                    shouldBlock = true;
                                    detectedLang = `Cyrylica (${cyrillicMatches.length} znaków)`;
                                }
                            } else if (otherNonLatinMatches.length > 3) {
                                if (latinWords.length < 3 || otherNonLatinMatches.length >= 8) {
                                    shouldBlock = true;
                                    detectedLang = `Pismo niełacińskie (${otherNonLatinMatches.length} znaków)`;
                                }
                            } else if (turkishCharMatches.length > 0 && !hasPolishDiacritics) {
                                shouldBlock = true;
                                detectedLang = `Turecki (litery: ${[...new Set(turkishCharMatches)].join(', ')})`;
                            } else if (foreignAccentWords.length >= 2 || (foreignAccentWords.length === 1 && latinWords.length <= 4)) {
                                // Ochrona przed czeskimi/niemieckimi bardzo krótkimi tytułami (wymaga tylko jednego dziwnego znaku przy max 4 słowach)
                                shouldBlock = true;
                                detectedLang = `Obce znaki diakrytyczne (${foreignAccentWords.length} słow.)`;
                            }

                            if (shouldBlock) {
                                hideReason = `Tytuł obcojęzyczny: ${detectedLang}`;
                            }
                        }

                        // Natywny analizator leksykalny dla języków europejskich i światowych w alfabecie łacińskim
                        if (!hideReason && !hasPolishDiacritics && titleText.length >= 6) {
                            const lexicalResult = detectLatinLanguageLexically(normalizedTitle, titleText);
                            if (lexicalResult) {
                                if (allowScandinavian && lexicalResult.isScandinavian) {
                                    // Nie blokuj, jeśli użytkownik zezwolił na języki skandynawskie
                                } else {
                                    hideReason = `Tytuł obcojęzyczny: ${lexicalResult.name} (${lexicalResult.hits} znaczników)`;
                                }
                            }
                        }

                        // Opcjonalne uzupełnienie przez Chrome AI (gdy dostępne)
                        if (!hideReason && useChromeAi && aiDetectorInstance && titleText.length >= 12) {
                            const cachedAi = aiDetectionCache.get(titleText);
                            if (cachedAi) {
                                const isScandicAi = allowScandinavian && scandicLanguageCodes.has(cachedAi.detectedLanguage);
                                if (!isScandicAi && cachedAi.detectedLanguage !== 'pl' && cachedAi.detectedLanguage !== 'en' && cachedAi.confidence >= 0.70) {
                                    const lName = langNames[cachedAi.detectedLanguage] || cachedAi.detectedLanguage.toUpperCase();
                                    hideReason = `Tytuł obcojęzyczny: ${lName} (AI: ${Math.round(cachedAi.confidence * 100)}%)`;
                                }
                            } else {
                                processAiDetection(item, titleText, rank);
                            }
                        }
                    }
                }

                if (!hideReason) {
                    if (rank === -1) {
                        hideReason = 'Brak informacji o dacie publikacji';
                    } else if (rank === -2) {
                        if (activeKeywords.some(kw => kw.toLowerCase() === 'sponsorowane')) {
                            hideReason = 'Film sponsorowany / Reklama';
                        }
                    } else {
                        const matchedKw = activeKeywords.find(kw => {
                            if (kw.toLowerCase() === 'sponsorowane') return false;

                            const exactMatch = kw.match(/^(\d+)([\p{L}\.]+)$/iu);

                            if (exactMatch) {
                                const num = exactMatch[1];
                                const letter = exactMatch[2].replace(/\./g, '\\.');
                                const regex = new RegExp(`(?:^|[^\\p{L}\\d])${num}\\s*${letter}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
                                return regex.test(fullText);
                            }

                            const safeKw = kw.replace(/\./g, '\\.');
                            const shortFormats = ['s', 'm', 'h', 'd', 'w', 'mo', 'y', 'sek', 'min', 'godz'];

                            if (shortFormats.includes(kw.toLowerCase())) {
                                const exactNumbersForThisKw = keywords
                                    .map(k => k.match(/^(\d+)([\p{L}\.]+)$/iu))
                                    .filter(m => m && m[2].toLowerCase() === kw.toLowerCase())
                                    .map(m => m[1]);

                                let excludePattern = '';
                                if (exactNumbersForThisKw.length > 0) {
                                    excludePattern = `(?!(?:${exactNumbersForThisKw.join('|')})\\s*${safeKw}(?![\\p{L}]))`;
                                }

                                const regex = new RegExp(`(?:^|[^\\p{L}\\d])${excludePattern}(\\d+)\\s*${safeKw}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
                                return regex.test(fullText);
                            } else {
                                const regex = new RegExp(`(?:^|[^\\p{L}])${safeKw}(?![\\p{L}])`, 'iu');
                                return regex.test(fullText);
                            }
                        });

                        if (matchedKw) {
                            hideReason = `Wskaźnik czasu: "${matchedKw}"`;
                        }
                    }
                }

                if (hideReason) {
                    hiddenCount++;
                }

                applyHidingState(item, hideReason, rank);
            });

            updateBadgeUI(hiddenCount);
        } catch (err) {
            addLog('ERROR', 'Błąd w processYouTubeHider', err.message);
        }
    }

        saveSettings();

    function injectBadge() {
        try {
            const mastheadButtons = document.querySelector('#buttons.ytd-masthead') || document.querySelector('#end #buttons');
            if (!mastheadButtons) return;
            if (document.getElementById('yt-hider-badge')) return;

            const badgeBtn = document.createElement('div');
            badgeBtn.id = 'yt-hider-badge';
            badgeBtn.style.display = 'flex';
            badgeBtn.style.alignItems = 'center';
            badgeBtn.style.backgroundColor = '#0f0f0f';
            badgeBtn.style.border = '1px solid #333';
            badgeBtn.style.borderRadius = '20px';
            badgeBtn.style.padding = '0 12px';
            badgeBtn.style.height = '32px';
            badgeBtn.style.marginRight = '8px';
            badgeBtn.style.cursor = 'pointer';
            badgeBtn.style.color = '#ffffff';
            badgeBtn.style.fontWeight = 'bold';
            badgeBtn.style.fontSize = '14px';

            const iconSpan = document.createElement('span');
            iconSpan.id = 'yt-hider-icon';
            iconSpan.textContent = '👁';
            iconSpan.style.color = '#ff8c00';
            iconSpan.style.marginRight = '6px';
            iconSpan.style.fontSize = '16px';

            const countText = document.createElement('span');
            countText.id = 'yt-hider-count';
            countText.textContent = '0';

            badgeBtn.appendChild(iconSpan);
            badgeBtn.appendChild(countText);

            badgeBtn.addEventListener('click', () => {
                const modal = document.getElementById('yt-hider-modal');
                if (modal) modal.style.display = modal.style.display === 'none' ? 'block' : 'none';
            });

            if (mastheadButtons.firstChild) {
                mastheadButtons.insertBefore(badgeBtn, mastheadButtons.firstChild);
            } else {
                mastheadButtons.appendChild(badgeBtn);
            }

            updateBadgeUI(0);
            addLog('INFO', 'injectBadge: Pomyślnie dodano plakietkę z okiem');
        } catch (err) {
            addLog('ERROR', 'Błąd w injectBadge', err.message);
        }
    }

    let hiderTimer;
    const observer = new MutationObserver((mutations) => {
        if (mutations.some(m => m.addedNodes.length > 0)) {
            clearTimeout(hiderTimer);
            hiderTimer = setTimeout(processYouTubeHider, 300);
        }
    });

    let currentContainer = null;
    function init() {
        if (!document.body) {
            setTimeout(init, 50);
            return;
        }

        addLog('INFO', 'init: Uruchamianie inicjalizacji');

        try {
            injectUI();
        } catch (e) {
            addLog('ERROR', 'Błąd w injectUI', e.message);
        }

        try {
            injectBadge();
        } catch (e) {
            addLog('ERROR', 'Błąd w injectBadge', e.message);
        }

        initChromeAiDetector();

        try {
            const container = document.querySelector('ytd-rich-grid-renderer > #contents') || document.querySelector('ytd-rich-grid-renderer #contents') || document.querySelector('#contents');
            if (container && currentContainer !== container) {
                if (currentContainer) observer.disconnect();
                currentContainer = container;
                observer.observe(container, { childList: true, subtree: true });
                addLog('INFO', 'init: Podpięto obserwatora pod kontener filmów');
            }

            if (container) {
                processYouTubeHider();
            } else {
                addLog('WARN', 'init: Brak kontenera filmów, ponawiam za 1s');
                setTimeout(init, 1000);
            }
        } catch (e) {
            addLog('ERROR', 'Błąd w init przy podpinaniu kontenera', e.message);
        }
    }

    init();
    window.addEventListener('yt-navigate-finish', () => {
        addLog('INFO', 'yt-navigate-finish: Zdarzenie nawigacji YouTube');
        injectBadge();
        processYouTubeHider();
    });
