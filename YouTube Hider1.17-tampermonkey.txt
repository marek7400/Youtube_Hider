    function getChannelNames(item) {
        const names = [];
        if (!item) return names;

        try {
            // 1. Wyciąganie tekstów z bezpośrednich linków (często zawiera @handle)
            const allLinks = item.querySelectorAll('a');
            allLinks.forEach(link => {
                const href = link.getAttribute('href') || '';
                if (href.startsWith('/@') || href.startsWith('/channel/') || href.startsWith('/c/') || href.startsWith('/user/')) {
                    if (href.startsWith('/@')) {
                        const handle = href.split('/')[1].split('?')[0];
                        names.push(handle);
                    }
                    if (link.textContent) names.push(link.textContent.replace(/\s+/g, ' '));
                    if (link.getAttribute('title')) names.push(link.getAttribute('title'));
                }
                if (link.id === 'avatar-link' && link.getAttribute('title')) {
                    names.push(link.getAttribute('title'));
                }
            });

            // 2. Łapanie klasycznych kontenerów nazw - łapie również collaby z tekstem pomiędzy
            const textNodes = item.querySelectorAll('ytd-channel-name, #channel-name, .ytd-channel-name, .yt-channel-name-view-model');
            textNodes.forEach(node => {
                if (node.getAttribute('title')) names.push(node.getAttribute('title'));
                if (node.textContent) names.push(node.textContent.replace(/\s+/g, ' '));
            });

            // 3. Łapanie klasycznych obrazków po atrybucie ALT
            const avatars = item.querySelectorAll('img');
            avatars.forEach(img => {
                if (img.getAttribute('alt')) names.push(img.getAttribute('alt'));
            });

            // 4. Łapanie nowych kształtów awatarów wprowadzanych przez yt-channel-name-view-model
            const avatarShapes = item.querySelectorAll('yt-avatar-shape');
            avatarShapes.forEach(shape => {
                const tooltip = shape.getAttribute('data-tooltip-text');
                if (tooltip) names.push(tooltip);
                const ariaLabel = shape.getAttribute('aria-label');
                if (ariaLabel) names.push(ariaLabel);
            });

        } catch (e) {
            addLog('ERROR', 'Błąd w getChannelNames', e.message);
        }

        // Filtrowanie, usuwanie duplikatów i znaków białych na końcach
        let cleanedNames = [];
        names.forEach(n => {
            let trimmed = n.trim();
            // Usuwanie ewentualnych niewidzialnych znaków (LRM) potrafiących psuć parsowanie
            trimmed = trimmed.replace(/[\u200B-\u200D\uFEFF\u200E\u200F]/g, '');

            if (trimmed.length > 0) {
                cleanedNames.push(trimmed);

                // Rozdzielanie wariantów typu "Kanał i jeszcze 2" lub "Channel and 2 more" (collabs)
                const collabMatch = trimmed.match(/^(.*?)\s+(?:i jeszcze \d+|and \d+ more)$/i);
                if (collabMatch && collabMatch[1]) {
                    cleanedNames.push(collabMatch[1].trim());
                }
            }
        });

        return [...new Set(cleanedNames)];
    }

    function getBestChannelName(item) {
        const names = getChannelNames(item);
        if (!names || names.length === 0) return null;
        
        // Priorytetyzujemy unikalny @handle jako idealny cel zablokowania w menu podglądu (omija on problemy tekstowe całkowicie)
        const handle = names.find(n => n.startsWith('@'));
        if (handle) return handle;

        // Jeżeli brak handle'a, szukamy najkrótszej nazwy (unikamy złączonych z kolaboracji, typu "WuWunio i jeszcze 2")
        const cleanNames = names.filter(n => !n.startsWith('@'));
        if (cleanNames.length > 0) {
            return cleanNames.reduce((a, b) => a.length <= b.length ? a : b);
        }

        return names[0];
    }

                        saveSettings();

                        const ta = document.getElementById('yt-hider-blocked-channels');

                }
            });

            targetContainer.style.setProperty('position', 'relative', 'important');
            targetContainer.appendChild(blockBtn);
        } catch (err) {
            addLog('ERROR', 'Błąd w injectBlockButton', err.message);
        }
    }

    function calculateRankAndGetTime(item) {
        if (item.querySelector('#ad-badge, .ytd-ad-slot-renderer') || item.textContent.toLowerCase().includes('sponsorowane') || item.textContent.toLowerCase().includes('sponsored')) {
            return { rank: -2, fullText: "sponsorowane" };
        }

        let flatMetadata = "";
        const metadataNodes = item.querySelectorAll('#metadata-line, .yt-core-attributed-string, .yt-lockup-metadata-view-model__metadata, .yt-content-metadata-view-model__metadata-row, #metadata');

        metadataNodes.forEach(node => {
            flatMetadata += " " + node.textContent.replace(/\s+/g, ' ');
        });

        flatMetadata = flatMetadata.toLowerCase();

        if (!flatMetadata.includes('temu') && !flatMetadata.includes('ago')) {
            const lines = (item.innerText || item.textContent || "").toLowerCase().split(/\r?\n/);
            for (const line of lines) {
                if (line.includes('temu') || line.includes('ago')) {
                    flatMetadata += " " + line.replace(/\s+/g, ' ');
                }
            }
        }

        const sortedUnits = Object.keys(timeUnits).sort((a, b) => b.length - a.length);

        for (const unit of sortedUnits) {
            const safeUnit = unit.replace(/\./g, '\\.');
            const regex = new RegExp(`(?:^|[^\\p{L}])(\\d+)\\s*${safeUnit}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
            const match = flatMetadata.match(regex);
            if (match) {
                return { rank: parseInt(match[1], 10) * timeUnits[unit], fullText: flatMetadata };
            }
        }

        for (const unit of sortedUnits) {
            const safeUnit = unit.replace(/\./g, '\\.');
            const regex = new RegExp(`(?:^|[^\\p{L}])${safeUnit}(?=[^\\p{L}]|$).{0,20}(?:temu|ago)`, 'iu');
            const match = flatMetadata.match(regex);
            if (match) {
                return { rank: 1 * timeUnits[unit], fullText: flatMetadata };
            }
        }

        return { rank: -1, fullText: flatMetadata };
    }

    function processYouTubeHider() {
        try {
            const gridContainer = document.querySelector('ytd-rich-grid-renderer > #contents') || document.querySelector('ytd-rich-grid-renderer #contents');
            if (!gridContainer) {
                return;
            }

            const items = gridContainer.querySelectorAll('ytd-rich-item-renderer, ytd-rich-section-renderer');
            let hiddenCount = 0;

            if (!pluginEnabled) {
                document.documentElement.classList.remove('youtube-hider-active');
                gridContainer.style.removeProperty('display');
                gridContainer.style.removeProperty('flex-wrap');
                items.forEach(item => {
                    item.style.removeProperty('display');
                    item.style.removeProperty('order');
                    item.style.removeProperty('width');
                    item.style.removeProperty('margin');
                    removeOverlay(item);
                    const btn = item.querySelector('.yt-hider-block-ch-btn');
                    if (btn) btn.remove();
                });
                updateBadgeUI(0);
                return;
            }

            document.documentElement.classList.add('youtube-hider-active');
            gridContainer.style.setProperty('display', 'flex', 'important');
            gridContainer.style.setProperty('flex-wrap', 'wrap', 'important');

            const activeKeywords = Object.keys(keywordSettings).filter(k => keywordSettings[k]);
            const customWordsArr = customBlockedWords.split(',').map(w => w.trim().toLowerCase()).filter(w => w.length > 0);

            items.forEach(item => {
                injectBlockButton(item);

                if (item.dataset.ytHiderUnblocked === "true") {
                    item.style.setProperty('display', 'block', 'important');
                    removeOverlay(item);

                    const { rank } = calculateRankAndGetTime(item);
                    const orderVal = rank >= 0 ? rank : 999999;
                    item.style.setProperty('order', orderVal, 'important');

                    if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                        item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                        item.style.setProperty('margin', '0 8px 24px 8px', 'important');
                    } else {
                        item.style.setProperty('width', '100%', 'important');
                        item.style.setProperty('margin', '0 0 24px 0', 'important');
                    }
                    return;
                }

                if (item.querySelector('a[href^="/shorts/"]')) {
                    item.style.setProperty('display', 'none', 'important');
                    return;
                }

                const { rank, fullText } = calculateRankAndGetTime(item);
                let hideReason = null;
                let titleText = '';

                const titleNodes = item.querySelectorAll('#video-title, #video-title-link');
                if (titleNodes.length > 0) {
                    titleNodes.forEach(node => { titleText += ' ' + node.textContent; });
                } else {
                    const h3Node = item.querySelector('h3');
                    if (h3Node) titleText = h3Node.textContent;
                }
                if (titleText) titleText = titleText.trim();

                const itemChannelNames = getChannelNames(item);

                // 1. Czarna lista kanałów
                if (!hideReason && blockedChannels.length > 0 && itemChannelNames.length > 0) {
                    const isBlocked = itemChannelNames.some(extracted =>
                        blockedChannels.some(blocked => blocked.toLowerCase() === extracted.toLowerCase())
                    );
                    if (isBlocked) {
                        hideReason = 'Block reason: Zablokowany kanał';
                    }
                }

                // 2. Własne słowa w tytule i nazwie kanału
                if (!hideReason && customWordsArr.length > 0) {
                    const lowerTitle = titleText ? titleText.toLowerCase() : '';
                    const matchedWord = customWordsArr.find(word => {
                        return lowerTitle.includes(word) || itemChannelNames.some(ch => ch.toLowerCase().includes(word));
                    });
                    if (matchedWord) {
                        hideReason = `Block reason: Custom word ('${matchedWord}')`;
                    }
                }

                // 3. Blokada po języku
                if (blockNonPlEn && !hideReason) {
                    const isChannelAllowed = itemChannelNames.some(extracted =>
                        allowedChannels.some(allowed => allowed.toLowerCase() === extracted.toLowerCase())
                    );

                    if (!isChannelAllowed && titleText) {
                        const match = titleText.match(foreignRegex);
                        if (match) {
                            hideReason = `Block reason: Non-PL/EN title (detected: '${match[0]}')`;
                        }
                    }
                }

                // 4. Filtry czasu i sponsorowane
                if (!hideReason) {
                    if (rank === -1) {
                        hideReason = 'Block reason: No valid metadata';
                    } else if (rank === -2) {
                        if (activeKeywords.some(kw => kw.toLowerCase() === 'sponsorowane')) {
                            hideReason = 'Block reason: Sponsored';
                        }
                    } else {
                        const matchedKw = activeKeywords.find(kw => {
                            if (kw.toLowerCase() === 'sponsorowane') return false;

                            const exactMatch = kw.match(/^(\d+)([\p{L}\.]+)$/iu);

                            if (exactMatch) {
                                const num = exactMatch[1];
                                const letter = exactMatch[2].replace(/\./g, '\\.');
                                const regex = new RegExp(`(?:^|[^\\p{L}\\d])${num}\\s*${letter}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
                                return regex.test(fullText);
                            }

                            const safeKw = kw.replace(/\./g, '\\.');
                            const shortFormats = ['s', 'm', 'h', 'd', 'w', 'mo', 'y', 'sek', 'min', 'godz'];

                            if (shortFormats.includes(kw.toLowerCase())) {
                                const exactNumbersForThisKw = keywords
                                    .map(k => k.match(/^(\d+)([\p{L}\.]+)$/iu))
                                    .filter(m => m && m[2].toLowerCase() === kw.toLowerCase())
                                    .map(m => m[1]);

                                let excludePattern = '';
                                if (exactNumbersForThisKw.length > 0) {
                                    excludePattern = `(?!(?:${exactNumbersForThisKw.join('|')})\\s*${safeKw}(?![\\p{L}]))`;
                                }

                                const regex = new RegExp(`(?:^|[^\\p{L}\\d])${excludePattern}(\\d+)\\s*${safeKw}(?![\\p{L}]|\\s*(?:views|wyświetle|ogląda|watching))`, 'iu');
                                return regex.test(fullText);
                            } else {
                                const regex = new RegExp(`(?:^|[^\\p{L}])${safeKw}(?![\\p{L}])`, 'iu');
                                return regex.test(fullText);
                            }
                        });

                        if (matchedKw) {
                            hideReason = `Block reason: ${matchedKw}`;
                        }
                    }
                }

                if (hideReason) {
                    hiddenCount++;

                    const isLanguageBlock = hideReason.startsWith('Block reason: Non-PL/EN title');

                    if (isLanguageBlock && showOverlays) {
                        item.style.setProperty('display', 'block', 'important');
                        item.style.setProperty('order', '9999999', 'important');

                        if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                            item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                            item.style.setProperty('margin', '0 8px 24px 8px', 'important');
                        } else {
                            item.style.setProperty('width', '100%', 'important');
                            item.style.setProperty('margin', '0 0 24px 0', 'important');
                        }
                        applyOverlay(item, hideReason);
                    } else {
                        item.style.setProperty('display', 'none', 'important');
                        removeOverlay(item);
                    }
                } else {
                    item.style.setProperty('display', 'block', 'important');
                    removeOverlay(item);

                    const orderVal = rank >= 0 ? rank : 999999;
                    item.style.setProperty('order', orderVal, 'important');

                    if (item.tagName.toLowerCase() === 'ytd-rich-item-renderer') {
                        item.style.setProperty('width', 'calc(25% - 16px)', 'important');
                        item.style.setProperty('margin', '0 8px 24px 8px', 'important');
                    } else {
                        item.style.setProperty('width', '100%', 'important');
                        item.style.setProperty('margin', '0 0 24px 0', 'important');
                    }
                }
            });

            updateBadgeUI(hiddenCount);
        } catch (err) {
            addLog('ERROR', 'Błąd w processYouTubeHider', err.message);
        }
    }

    function handleKeywordChange(e) {
        const kw = e.target.getAttribute('data-keyword');
        keywordSettings[kw] = e.target.checked;
        saveSettings();
    }

    let currentContainer = null;
    function init() {
        if (!document.body) {
            setTimeout(init, 50);
            return;
        }

        addLog('INFO', 'init: Uruchamianie inicjalizacji');

        try {
            injectUI();
        } catch (e) {
            addLog('ERROR', 'Błąd w injectUI', e.message);
        }

        try {
            injectBadge();
        } catch (e) {
            addLog('ERROR', 'Błąd w injectBadge', e.message);
        }

        try {
            const container = document.querySelector('ytd-rich-grid-renderer > #contents') || document.querySelector('ytd-rich-grid-renderer #contents');
            if (container && currentContainer !== container) {
                if (currentContainer) observer.disconnect();
                currentContainer = container;
                observer.observe(container, { childList: true });
                addLog('INFO', 'init: Podpięto obserwatora pod kontener filmów');
            }

            if (container) {
                processYouTubeHider();
            } else {
                addLog('WARN', 'init: Brak kontenera filmów, ponawiam za 1s');
                setTimeout(init, 1000);
            }
        } catch (e) {
            addLog('ERROR', 'Błąd w init przy podpinaniu kontenera', e.message);
        }
    }

    init();
    window.addEventListener('yt-navigate-finish', () => {
        addLog('INFO', 'yt-navigate-finish: Zdarzenie nawigacji YouTube');
        injectBadge();
        processYouTubeHider();
    });
